#![no_std]
mod sealed {
    pub struct Token;
    pub trait Sealed {
        #[cfg(not(feature = "t1"))]
        fn test(&self, t: Token);
        #[cfg(feature = "t1")]
        fn test(&self);
        const N: u32;
        type A;
    }
}
pub trait Downcast: sealed::Sealed {}
pub struct S;
impl sealed::Sealed for S {
    #[cfg(not(feature = "t1"))]
    fn test(&self, _t: sealed::Token) {}
    #[cfg(feature = "t1")]
    fn test(&self) {}
    const N: u32 = 1;
    type A = u8;
}
impl Downcast for S {}
